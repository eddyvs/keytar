    Those are aubio's frequency estimates in hertz, one estimate per audio block. They aren't note names, and individual estimates can jump when the sound is quiet, noisy, or between notes.

    Around 375-387 Hz: roughly F#4, drifting toward G4. F#4 is about 370 Hz; G4 is about 392 Hz.
    Around 56-65 Hz: roughly the A1-C2 range. If you're playing higher notes, these may be low-frequency noise or incorrect estimates.
    7,659 Hz: unusually high for a musical fundamental; likely a transient, noise, or a pitch-tracking error.
    88,200 Hz: not a valid pitch from a 44.1 kHz recording. The Nyquist limit is 22,050 Hz, so treat this as an invalid estimate.
    Try printing the note and aubio confidence together, and only accept estimates inside the range you expect.

    def onStep(app):
        frame = audioStream.read(CHUNK, exception_on_overflow=False)
        samples = np.frombuffer(frame, dtype=np.int16).astype(np.float32) / 32768.0
        frequency = float(pitch_detector(samples)[0])

        if frequency < 90 or frequency > 1200:
            app.currentNote = "--"
            app.currentKey = None
            app.previousKeyBind = None
            keyUpAll()
            return

            keyUpAll()

                    "down": (0, 10),
                    "left": (-10, 0),
                    "right": (10, 0),

            else:
                pydirectinput.keyDown(keyBind)
        else:
            keyUpAll()
            if keyBind not in ["click", "up", "down", "left", "right"]:
                pydirectinput.keyDown(keyBind)
            app.previousKeyBind = keyBind

    def queueDisplayState(note, keybind, morseMode, morseSymbols, lastDecodedLetter):
        note_queue.append((note, keybind, morseMode,
                           "".join(morseSymbols), lastDecodedLetter))


    def finishMorseLetter(morseSymbols):
        letter = MORSE_CODES.get("".join(morseSymbols))
        if letter is None:
            return "?"
        pydirectinput.press(letter)
        return letter


    def keybindLoop():
        previousKeybind = None
        previousNote = None
        morseMode = False

        lastDecodedLetter = ""

            frequency = float(pitch_detector(samples)[0])
            now = time.monotonic()

            if frequency > 1200 or frequency < 90:
                if previousNote is not None:
                    previousNote = None
                    if morseMode:
                        silenceStarted = now
                    else:

                        previousKeybind = None
                    queueDisplayState(None, None, morseMode, morseSymbols,
                                      lastDecodedLetter)

                if (morseMode and morseSymbols and silenceStarted is not None
                        and now - silenceStarted >= MORSE_LETTER_GAP):
